\chapterdivider{3}{Natural and pseudo-arclength continuation}
  {How can a branch follower pass a point where the physical parameter turns around?}
  {Featured application: predictor--corrector continuation through a saddle-node fold}

\begin{frame}{Natural continuation fixes the next parameter}
\[
  p_{\mathrm{pred}}
    =p_k+\operatorname{sign}(p_{\mathrm{end}}-p_k)\,\Delta s,
  \qquad
  F(\state,p_{\mathrm{pred}})=0
  \quad\text{with $p_{\mathrm{pred}}$ fixed.}
\]
\begin{block}{When it works well}
Natural continuation is simple and inexpensive when the branch is locally a
single-valued function $\state(p)$ and folds are not part of the question.
\end{block}
\end{frame}

\begin{frame}{Pseudo-arclength changes the coordinate}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{alertblock}{Natural continuation}
The physical parameter is prescribed. Only the state is corrected, so the
method inherits every failure of $p$ as a local coordinate.
\end{alertblock}

\column{.48\textwidth}
\begin{block}{Pseudo-arclength continuation}
State and parameter are corrected together. A geometric progress condition
selects the next point along the curve.
\end{block}
\end{columns}

\vfill
\centering
The artificial coordinate $s$ keeps increasing even when $p$ turns around.
\end{frame}

\begin{frame}{One extra equation selects the next point}
The corrector solves $n+1$ equations for $(\state,p)$:
\[
\begin{cases}
F(\state,p)=0,\\[1mm]
g(\state,p)=
(\state-\state_k)^T\dot{\state}_k
+(p-p_k)\dot p_k-\Delta s=0.
\end{cases}
\]
\begin{block}{Two geometric roles}
$F=0$ places the point on the branch; $g=0$ places it on a hyperplane normal
to the previous unit tangent. Their local intersection isolates one point.
\end{block}
\end{frame}

\begin{frame}{A bordered solve computes the next tangent}
\[
\begin{bmatrix}
F_{\state} & F_p\\
\tangent_{k-1}^{T}
\end{bmatrix}
\tangent_k=
\begin{bmatrix}\bm{0}\\1\end{bmatrix},
\qquad
\tangent_k\leftarrow\frac{\tangent_k}{\|\tangent_k\|_2},
\qquad
\tangent_k^T\tangent_{k-1}>0.
\]
\begin{block}{Why the border matters}
The residual derivative fixes the tangent line; the final row fixes scale and
orientation. Sign alignment keeps successive predictors moving consistently.
\end{block}
\end{frame}

\begin{frame}{The Newton corrector uses the same geometry}
One augmented Newton iteration solves
\[
\underbrace{\begin{bmatrix}
F_{\state} & F_p\\
\dot{\state}_k^T & \dot p_k
\end{bmatrix}}_{\text{bordered Jacobian}}
\begin{bmatrix}\delta\state\\\delta p\end{bmatrix}
=-
\begin{bmatrix}F(\state,p)\\g(\state,p)\end{bmatrix},
\qquad
(\state,p)\leftarrow(\state,p)+(\delta\state,\delta p).
\]
\begin{alertblock}{The fold is no longer singular to the corrector}
Although $F_{\state}$ is singular at an ordinary fold, the full bordered
matrix can remain nonsingular because the parameter direction is included.
\end{alertblock}
\end{frame}

\begin{frame}{What to remember}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{Geometric idea}
Predict along the branch tangent, then correct on a transverse hyperplane
while allowing both state and parameter to move.
\end{block}

\column{.48\textwidth}
\begin{block}{Numerical consequence}
The augmented equation and bordered systems replace a singular fixed-$p$
correction with a regular local curve coordinate.
\end{block}
\end{columns}

\vfill
\centering
\large\textcolor{JaxTeal}{The engine can now follow any finite-dimensional residual whose input and output shapes match.}
\end{frame}
