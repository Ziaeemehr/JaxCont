\begin{frame}{A cycle becomes three residual blocks}
Split normalized time $\tau\in[0,1]$ into \texttt{ntst} intervals and place
\texttt{ncol} Gauss--Legendre points inside each interval:
\[
  \frac{d\state}{d\tau}=T f(\state,p).
\]
\begin{center}
\begin{tikzpicture}[x=1.05cm,y=1cm,>=Latex]
  \draw[->,thick] (0,0)--(10.3,0) node[right] {$\tau$};
  \foreach \x/\lab in {0/0,2.5/{1/4},5/{1/2},7.5/{3/4},10/1} {
    \draw[very thick,JaxBlue] (\x,-.12)--(\x,.12);
    \node[below=2mm] at (\x,0) {$\lab$};
  }
  \foreach \base in {0,2.5,5,7.5} {
    \foreach \off in {.45,1.25,2.05}
      \fill[JaxOrange] (\base+\off,0) circle (2.3pt);
  }
  \node[above=4mm,JaxBlue] at (1.25,0) {mesh interval};
  \node[above=4mm,JaxOrange!90!black] at (6.25,0) {collocation points};
\end{tikzpicture}
\end{center}
\begin{enumerate}
  \item \textbf{Defects:} polynomial derivatives match $T f$ at collocation points.
  \item \textbf{Continuity:} each interval endpoint matches the next, with wrap-around.
  \item \textbf{Phase:} one scalar condition removes arbitrary time translation.
\end{enumerate}
\end{frame}

\begin{frame}{Why the phase condition is necessary}
For normalized phase $\tau,\phi\in[0,1)$, every shift
$\state((\tau+\phi)\bmod 1)$ traces the same geometric orbit.
\begin{center}
\begin{tikzpicture}[x=1.1cm,y=1.1cm,>=Latex]
  \draw[JaxBlue,very thick] (0,0) ellipse (2.3 and 1.25);
  \fill[JaxOrange] (2.3,0) circle (3pt) node[right] {$\phi=0$};
  \fill[JaxGreen!80!black] (0,1.25) circle (3pt) node[above] {$\phi=1/4$};
  \draw[->,thick,JaxOrange] (1.9,.75)
      arc[start angle=35,end angle=75,x radius=2.3,y radius=1.25];
  \node[align=center] at (0,-1.85) {same closed curve, different starting point};
\end{tikzpicture}
\end{center}
\begin{block}{Numerical consequence}
Without a phase condition, sliding all samples in time changes $U$ but not
the orbit. The collocation Jacobian therefore has a neutral direction; the
phase condition selects one representative of that equivalence class.
\end{block}
\end{frame}

\begin{alertblock}{Responsibility boundary}
The user supplies a coarse trajectory representing one cycle and a reasonable
period. The factory refines that supplied coarse cycle; it does not discover a
cycle by integrating from a Hopf point.
\end{alertblock}
\begin{block}{Interpolation boundary}

\begin{frame}{What is packed into a periodic branch state?}
For state dimension $n$, \texttt{ntst} intervals, and \texttt{ncol}
collocation points, each branch state has
\[
  n\,\texttt{ntst}\,(1+\texttt{ncol})+1
\]
entries.
\begin{center}
\begin{tikzpicture}[x=.85cm,y=.8cm]
  \draw[fill=JaxBlue!25] (0,0) rectangle (4.4,1);
  \draw[fill=JaxOrange!25] (4.4,0) rectangle (10.6,1);
  \draw[fill=JaxGreen!25] (10.6,0) rectangle (11.5,1);
  \node[align=center] at (2.2,.5) {mesh-point\\states};
  \node[align=center] at (7.5,.5) {collocation-point\\states};
  \node at (11.05,.5) {$T$};
\end{tikzpicture}
\end{center}
\begin{itemize}
  \item The final entry is the orbit period.
  \item Leading mesh states reshape to \texttt{(n\_valid, ntst, n)} for phase portraits.
  \item The full packed vector is the numerical unknown, not merely the physical state.
\end{itemize}
\begin{alertblock}{Interpretation rule}
\texttt{result.branch.states[:, 0]} is one component at one mesh point---not
automatically the orbit amplitude.
\end{alertblock}
\end{frame}

\begin{frame}{Floquet multipliers: stability of a cycle}
Perturb a point on a period-$T$ orbit and evolve the linearized perturbation
for one full period:
\[
  \dot\Phi(t)=f_{\state}(\state(t),p)\Phi(t),\qquad \Phi(0)=I.
\]
The eigenvalues $\mu_i$ of the monodromy matrix $\Phi(T)$ are the
\alert{Floquet multipliers}.
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Stable periodic orbit}
Ignore the one trivial multiplier nearest $+1$. Every remaining multiplier
must satisfy $|\mu_i|<1$.
\end{block}
\column{.48\textwidth}
\begin{alertblock}{Different from equilibria}
Equilibrium stability uses $\operatorname{Re}\lambda_i<0$; applying that rule
to Floquet multipliers is mathematically wrong.
\end{alertblock}
\end{columns}
\vspace{2mm}
JaxCont reconstructs the monodromy matrix from the collocation solution and
dispatches \texttt{compute\_stability=True} by \texttt{problem.kind}.
\end{frame}

\begin{frame}{Periodic-orbit numerics: practical boundaries}
\begin{itemize}
  \item \textbf{Fixed mesh:} \texttt{ntst} and \texttt{ncol} fix array shapes;
        v0.4.0 does not redistribute the mesh adaptively.
  \item \textbf{Large residual:} collocation expands the unknown, so dense
        Newton and eigenvalue solves become expensive quickly.
  \item \textbf{Float32 tolerance:} examples use \texttt{newton\_tol} around
        $10^{-5}$ or $10^{-4}$ depending on the system and mesh.
  \item \textbf{Guess quality:} the supplied trajectory must represent one
        cycle and carry a reasonable period estimate.
  \item \textbf{Event semantics:} Fold works on either branch kind; Hopf is
        equilibrium-only, while PD and NS are periodic-only.
\end{itemize}
\begin{block}{Static mesh metadata}
\texttt{Collocation.ntst} and \texttt{ncol} are static because they determine
array shapes and compilation. This makes the static/traced split explicit.
\end{block}
\end{frame}

\begin{frame}{What to remember}
\begin{enumerate}
  \item Collocation converts an entire cycle and its period into one finite
        nonlinear root with defect, continuity, and phase residuals.
  \item The user supplies a coarse cycle; \texttt{periodic\_orbit\_problem}
        resamples and refines it before continuation.
  \item Cycle stability is a Floquet-multiplier magnitude test after removing
        the trivial multiplier nearest $+1$.
  \item PD and NS events are unit-circle crossings, and Examples 08--09 check
        their detected locations against analytic answers.
\end{enumerate}
\vspace{3mm}
\begin{alertblock}{Scope boundary}
JaxCont continues a supplied/refined periodic orbit. Automatic cycle
discovery from a Hopf point, general BVP continuation, and connecting-orbit
continuation are not implemented. LPC/PD/NS detection remains experimental
until the strict \texttt{MC-LC-002} comparison passes.
\end{alertblock}
\end{frame}
