\begin{frame}{Choose the view from the scientific question}
\scriptsize
\centering

\setlength{\tabcolsep}{3.5pt}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{.20\textwidth}
                  >{\raggedright\arraybackslash}p{.34\textwidth}
                  >{\raggedright\arraybackslash}p{.38\textwidth}@{}}
\toprule
View & Answers & Does not establish\\
\midrule
Branch diagram
  & How solutions vary with parameter
  & Basins or full flow geometry\\
Eigenvalue/Floquet view
  & How local stability changes
  & Nonlinear long-time behavior\\
Branch-state projection
  & How stored branch states relate
  & A true phase portrait\\
2D phase plane
  & Nullclines, vector field, equilibria, and trajectories at one parameter
  & Higher-dimensional geometry or automatic branch discovery\\
\bottomrule
\end{tabular}
\vspace{3mm}
\begin{block}{Two complementary parameter choices}
Continuation traverses a branch while the parameter changes. A phase plane
freezes one parameter value and asks how states move there.
\end{block}
\end{frame}

\begin{block}{Compose only the evidence you need}
\texttt{plot\_phase\_plane} layers the 2D helpers in a readable order. Call a
helper directly when the question needs only nullclines, flow, equilibria, or
a trajectory.
\end{block}
\end{frame}

\vspace{3mm}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Practical recipe}
Choose the question, pass the corresponding data object, then label axes and
inspect stability/event markers before interpreting the picture.
\end{block}
\column{.48\textwidth}
\begin{alertblock}{Visualization is not simulation}
JaxCont does not own a time integrator. A trajectory layer either calls
SciPy's \texttt{solve\_ivp} with an explicit time span or draws a supplied
\texttt{(n\_steps, 2)} state array.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Example 12: compose the FitzHugh--Nagumo view}
\begin{lstlisting}
from jaxcont.viz import plot_phase_plane

result = jc.continuation(
    problem,
    p_span=(0.0, 1.0),
    settings=jc.ContinuationPar(ds=0.01, max_steps=400),
    events=[jc.Hopf()],
)

plot_phase_plane(
    problem, p=0.5,
    xlim=(-2.5, 2.5), ylim=(-1.0, 2.0),
    result=result,
    trajectories=[(jnp.array([-1.0, -0.5]),
                   (0.0, 200.0))],
)
\end{lstlisting}
\vspace{-3mm}
\begin{block}{What each input contributes}
\scriptsize
\texttt{problem} supplies the frozen vector field and state names;
\texttt{result} supplies continued equilibria and stability; the explicit
initial condition and time span request one trajectory overlay.
\end{block}
\end{frame}
